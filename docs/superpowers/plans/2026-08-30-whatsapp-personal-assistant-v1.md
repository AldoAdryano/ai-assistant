# WhatsApp Personal Assistant V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a zero-subscription, text-only personal assistant that receives WhatsApp Cloud API test messages, routes deterministic commands or Gemini-backed chat, persists notes/tasks/memory in Notion, and runs on Cloudflare Workers without a dedicated PC or VPS.

**Architecture:** A TypeScript Cloudflare Worker exposes `/health` and `/webhook`. The webhook verifies Meta signatures, allowlists one WhatsApp sender, suppresses repeated message IDs with Workers KV, acknowledges Meta quickly, then processes text through a deterministic parser/router; Gemini Interactions API is used only for chat or ambiguous classification, and Notion API `2026-03-11` is the persistent store.

**Tech Stack:** TypeScript, Cloudflare Workers + Workers KV, Wrangler v4, Vitest 4.1+ with `@cloudflare/vitest-plugin`, Meta WhatsApp Cloud API / Graph API v26.0, Gemini Interactions API v1 with `gemini-2.5-flash-lite`, Notion REST API `2026-03-11`.

**Spec:** `docs/superpowers/specs/2026-08-30-whatsapp-personal-assistant-v1-design.md`

## Global Constraints

- V1 supports WhatsApp text only; calendar, web research, PDFs, voice notes, images, groups, multiple users, long-term transcript storage, n8n, and background autonomous agents remain out of scope.
- Use Meta WhatsApp Cloud API test number and Graph API `v26.0`; do not use browser automation or unofficial WhatsApp libraries.
- Only `ALLOWED_WHATSAPP_NUMBER` may trigger assistant actions.
- Verify every webhook `POST` using `X-Hub-Signature-256` and `META_APP_SECRET` before parsing or acting.
- Store only processed WhatsApp message IDs in `DEDUP_KV`, with a 24-hour TTL (`86400` seconds).
- Use `gemini-2.5-flash-lite` by default, configurable through non-secret Worker variable `GEMINI_MODEL`.
- Use Gemini Interactions API in stateless mode with `store=false`; do not rely on Google-hosted interaction history.
- Use Notion API version `2026-03-11` and `data_source_id` parents/queries.
- Do not log access tokens, API keys, full webhook payloads, or full Notion content.
- Never confirm a Notion write unless the Notion API returned success.
- Deterministic commands bypass Gemini whenever they can be safely parsed.
- All async work must be awaited or passed to `ctx.waitUntil()`.
- No secrets may be committed; `.dev.vars` is gitignored.
- WhatsApp send retries are limited to one retry for HTTP 429 or 5xx responses.

Create `src/parser.ts`:

```ts
import type { ParsedIntent, Priority } from "./types";

function clean(value: string): string {
  return value.replace(/^\s*[:,-]?\s*/, "").replace(/\s+/g, " ").trim();
}

function parsePriority(text: string): { text: string; priority: Priority } {
  const patterns: Array<[RegExp, Priority]> = [
    [/(?:,?\s*)prioritas\s+(?:tinggi|high)\b/i, "High"],
    [/(?:,?\s*)prioritas\s+(?:rendah|low)\b/i, "Low"],
    [/(?:,?\s*)prioritas\s+(?:sedang|medium)\b/i, "Medium"],
  ];

  for (const [pattern, priority] of patterns) {
    if (pattern.test(text)) {
      return { text: clean(text.replace(pattern, "")), priority };
    }
  }

  return { text: clean(text), priority: "Medium" };
}

function parseDue(text: string): { text: string; due?: string } {
  const iso = text.match(/(?:deadline|jatuh tempo)\s+(\d{4}-\d{2}-\d{2})\b/i);
  if (iso?.[1]) {
    return { text: clean(text.replace(iso[0], "")), due: iso[1] };
  }

  const local = text.match(/(?:deadline|jatuh tempo)\s+(\d{2})\/(\d{2})\/(\d{4})\b/i);
  if (local?.[1] && local[2] && local[3]) {
    return {
      text: clean(text.replace(local[0], "")),
      due: `${local[3]}-${local[2]}-${local[1]}`,
    };
  }

  return { text: clean(text) };
}

export function parseDeterministicIntent(text: string, _now = new Date()): ParsedIntent | null {
  const raw = text.trim();
  const normalized = raw.toLocaleLowerCase("id-ID");

  if (/^(bantuan|help|menu)\s*[.!?]*$/i.test(raw)) {
    return { type: "HELP", raw };
  }

  const recall = raw.match(/^apa yang kamu ingat tentang\s+(.+?)[?!.]*$/i);
  if (recall?.[1]) {
    return { type: "RECALL", raw, query: clean(recall[1]) };
  }

  const remember = raw.match(/^ingat bahwa\s+(.+)$/i);
  if (remember?.[1]) {
    const statement = clean(remember[1]);
    const split = statement.match(/^(.+?)\s+adalah\s+(.+)$/i);
    return {
      type: "REMEMBER",
      raw,
      key: clean(split?.[1] ?? statement),
      value: clean(split?.[2] ?? statement),
      category: "Other",
    };
  }

  if (/^(apa tugas saya|daftar tugas)(\s+saya)?\s*[?!.]*$/i.test(raw)) {
    return { type: "LIST_TASKS", raw };
  }

  const task = raw.match(/^tambah tugas\s+(.+)$/i);
  if (task?.[1]) {
    const due = parseDue(task[1]);
    const priority = parsePriority(due.text);
    return {
      type: "ADD_TASK",
      raw,
      text: priority.text,
      priority: priority.priority,
      ...(due.due ? { due: due.due } : {}),
    };
  }

  const idea = raw.match(/^(?:catat ide|simpan ide)\s*:?[ ]*(.+)$/i);
  if (idea?.[1]) {
    return { type: "ADD_NOTE", raw, noteType: "Idea", text: clean(idea[1]) };
  }

  const note = raw.match(/^catat\s+(.+)$/i);
  if (note?.[1]) {
    return { type: "ADD_NOTE", raw, noteType: "Note", text: clean(note[1]) };
  }

  if (normalized === "") {
    return { type: "UNKNOWN", raw };
  }

  return null;
}
```

```ts
import { describe, expect, it } from "vitest";
import { verifyMetaSignature, verifyWebhookChallenge } from "../src/security";

async function sign(body: string, secret: string): Promise<string> {
  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );
  const bytes = new Uint8Array(await crypto.subtle.sign("HMAC", key, new TextEncoder().encode(body)));
  const hex = Array.from(bytes, (byte) => byte.toString(16).padStart(2, "0")).join("");
  return `sha256=${hex}`;
}

describe("Meta webhook security", () => {
  it("returns the challenge for a valid verification token", () => {
    const url = new URL(
      "https://example.test/webhook?hub.mode=subscribe&hub.verify_token=secret&hub.challenge=12345",
    );
    const response = verifyWebhookChallenge(url, "secret");
    expect(response.status).toBe(200);
  });

  it("rejects an invalid verification token", () => {
    const url = new URL(
      "https://example.test/webhook?hub.mode=subscribe&hub.verify_token=wrong&hub.challenge=12345",
    );
    expect(verifyWebhookChallenge(url, "secret").status).toBe(403);
  });

  it("accepts a valid HMAC signature", async () => {
    const body = '{"object":"whatsapp_business_account"}';
    const signature = await sign(body, "app-secret");
    await expect(verifyMetaSignature(body, signature, "app-secret")).resolves.toBe(true);
  });

  it("rejects missing or altered signatures", async () => {
    const body = '{"object":"whatsapp_business_account"}';
    await expect(verifyMetaSignature(body, null, "app-secret")).resolves.toBe(false);
    await expect(
      verifyMetaSignature(body, `sha256=${"00".repeat(32)}`, "app-secret"),
    ).resolves.toBe(false);
  });
});
```

```ts
function hexToBytes(hex: string): Uint8Array | null {
  if (!/^[0-9a-f]{64}$/i.test(hex)) return null;
  const bytes = new Uint8Array(32);

  }

export async function verifyMetaSignature(
  rawBody: string,
  signatureHeader: string | null,
  appSecret: string,
): Promise<boolean> {
  if (!signatureHeader?.startsWith("sha256=")) return false;
  const signature = hexToBytes(signatureHeader.slice("sha256=".length));
  if (!signature) return false;

  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(appSecret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["verify"],
  );

  return crypto.subtle.verify(
    "HMAC",
    key,
    signature,
    new TextEncoder().encode(rawBody),
  );
}
```

```ts
export function extractWhatsAppMessages(payload: unknown): IncomingWhatsAppMessage[] {
  if (typeof payload !== "object" || payload === null) return [];
  const root = payload as { entry?: unknown };
  if (!Array.isArray(root.entry)) return [];

  const output: IncomingWhatsAppMessage[] = [];

  for (const entry of root.entry) {
    if (typeof entry !== "object" || entry === null) continue;
    const changes = (entry as { changes?: unknown }).changes;
    if (!Array.isArray(changes)) continue;

    for (const change of changes) {
      if (typeof change !== "object" || change === null) continue;
      const value = (change as { value?: unknown }).value;
      if (typeof value !== "object" || value === null) continue;
      const messages = (value as { messages?: unknown }).messages;
      if (!Array.isArray(messages)) continue;

      for (const rawMessage of messages) {
        const message = rawMessage as MetaMessage;
        if (typeof message.id !== "string" || typeof message.from !== "string") continue;

        if (message.type === "text" && typeof message.text?.body === "string") {
          output.push({ id: message.id, from: message.from, kind: "text", text: message.text.body });
        } else {
          output.push({ id: message.id, from: message.from, kind: "unsupported" });
        }
      }
    }
  }

  return output;
}
```

```ts
function isTransient(status: number): boolean {
  return status === 429 || status >= 500;

export async function sendWhatsAppText(
  config: AppConfig,
  to: string,
  body: string,
  fetchImpl: typeof fetch = fetch,
): Promise<void> {
  const url = `https://graph.facebook.com/${config.graphApiVersion}/${config.metaPhoneNumberId}/messages`;
  const init: RequestInit = {
    method: "POST",
    headers: {
      Authorization: `Bearer ${config.metaAccessToken}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      messaging_product: "whatsapp",
      to,
      type: "text",
      text: { body },
    }),
  };

  for (let attempt = 1; attempt <= 2; attempt += 1) {
    const response = await fetchImpl(url, init);
    if (response.ok) return;

    if (attempt === 1 && isTransient(response.status)) {
      await scheduler.wait(300);
      continue;
    }

    throw new Error(`WhatsApp send failed with HTTP ${response.status}`);
  }
}
```

```ts
async function notionRequest<T>(
  config: AppConfig,
  path: string,
  init: RequestInit,
  fetchImpl: typeof fetch,
): Promise<T> {
  const response = await fetchImpl(`https://api.notion.com/v1${path}`, {
    ...init,
    headers: {
      Authorization: `Bearer ${config.notionApiKey}`,
      "Notion-Version": config.notionVersion,
      "Content-Type": "application/json",
      ...(init.headers ?? {}),
    },
  });

  if (!response.ok) {
    throw new Error(`Notion request failed with HTTP ${response.status}`);
  }

  return response.json<T>();
}

function titleText(property: any): string {
  return property?.title?.map((item: any) => item?.plain_text ?? "").join("").trim() ?? "";
}

function richText(property: any): string {
  return property?.rich_text?.map((item: any) => item?.plain_text ?? "").join("").trim() ?? "";
}

function textItems(text: string): Array<{ type: "text"; text: { content: string } }> {
  const items: Array<{ type: "text"; text: { content: string } }> = [];
  for (let start = 0; start < text.length; start += 2000) {
    items.push({ type: "text", text: { content: text.slice(start, start + 2000) } });
  }
  return items.length > 0 ? items : [{ type: "text", text: { content: "" } }];
}
```

```ts
export async function handleUserText(
  config: AppConfig,
  text: string,
  deps: RouterDeps = defaultDeps,
): Promise<string> {
  let intent: ParsedIntent | null = parseDeterministicIntent(text);

  if (!intent) {
    try {
      intent = await deps.classifyIntent(config, text);
    } catch {
      return "Layanan AI sedang tidak tersedia. Perintah catatan/tugas yang jelas masih dapat dicoba.";
    }
  }

  try {
    switch (intent.type) {
      case "ADD_NOTE":
        if (!intent.text) return "Isi catatan belum terbaca. Gunakan: Catat ...";
        await deps.createNote(config, { text: intent.text, noteType: intent.noteType ?? "Note" });
        return `Sudah saya simpan ke Inbox sebagai ${intent.noteType === "Idea" ? "ide" : "catatan"}.`;

      case "ADD_TASK":
        if (!intent.text) return "Isi tugas belum terbaca. Gunakan: Tambah tugas ...";
        await deps.createTask(config, {
          task: intent.text,
          priority: intent.priority ?? "Medium",
          ...(intent.due ? { due: intent.due } : {}),
        });
        return `Tugas sudah ditambahkan dengan prioritas ${intent.priority ?? "Medium"}.`;

      case "LIST_TASKS": { 
        const tasks = await deps.listActiveTasks(config);
        if (tasks.length === 0) return "Tidak ada tugas aktif di Notion.";
        return tasks.slice(0, 20).map((task, index) =>
          `${index + 1}. [${task.priority}] ${task.task}${task.due ? ` — ${task.due}` : ""}`,
        ).join("\n");
      }

      case "REMEMBER":
        if (!intent.key || !intent.value) return "Informasi yang harus diingat belum terbaca.";
        await deps.upsertMemory(config, {
          key: intent.key,
          value: intent.value,
          category: intent.category ?? "Other",
        });
        return "Sudah saya simpan ke Memory.";

      case "RECALL": { 
        if (!intent.query) return "Sebutkan hal yang ingin dicari di Memory.";
        const memories = await deps.recallMemory(config, intent.query);
        if (memories.length === 0) return "Saya belum menemukan Memory yang cocok.";
        return memories.map((memory, index) => `${index + 1}. ${memory.key}: ${memory.value}`).join("\n");
      }

      case "HELP":
        return HELP;

      case "CHAT": {
        const [tasks, memories] = await Promise.all([
          deps.listActiveTasks(config),
          deps.listMemoryContext(config),
        ]);
        return await deps.generateChatReply(config, text, { tasks, memories });
      }

      case "UNKNOWN":
      default:
        return HELP;
    }
  } catch {
    if (["ADD_NOTE", "ADD_TASK", "REMEMBER"].includes(intent.type)) {
      return "Penyimpanan ke Notion belum berhasil. Data belum saya nyatakan tersimpan.";
    }
    if (intent.type === "CHAT") {
      return "Layanan AI sedang tidak tersedia. Coba lagi nanti.";
    }
    return "Permintaan belum berhasil diproses. Coba lagi.";
  }
}
```

```ts
import { getConfig } from "./config";
import { handleUserText } from "./router";
import { verifyMetaSignature, verifyWebhookChallenge } from "./security";
import type { AppConfig, Env, IncomingWhatsAppMessage } from "./types";
import { extractWhatsAppMessages, sendWhatsAppText } from "./whatsapp";

const DEDUP_TTL_SECONDS = 86400;

export interface WorkerDeps {
  handleUserText: typeof handleUserText;
  sendWhatsAppText: typeof sendWhatsAppText;
}

const defaultDeps: WorkerDeps = { handleUserText, sendWhatsAppText };

async function processMessage(
  config: AppConfig,
  message: IncomingWhatsAppMessage,
  deps: WorkerDeps,
): Promise<void> {
  if (message.kind === "unsupported") {
    await deps.sendWhatsAppText(config, message.from, "V1 saat ini hanya mendukung pesan teks.");
    return;
  }

  const reply = await deps.handleUserText(config, message.text ?? "");
  await deps.sendWhatsAppText(config, message.from, reply);
}

export function createWorker(deps: WorkerDeps = defaultDeps): ExportedHandler<Env> {
  return {
    async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
      const url = new URL(request.url);

      if (request.method === "GET" && url.pathname === "/health") {
        return Response.json({ status: "ok" });
      }

      if (url.pathname !== "/webhook") {
        return new Response("Not found", { status: 404 });
      }

      const config = getConfig(env);

      if (request.method === "GET") {
        return verifyWebhookChallenge(url, config.metaVerifyToken);
      }

      if (request.method !== "POST") {
        return new Response("Method not allowed", { status: 405 });
      }

      const rawBody = await request.text();
      const signatureOk = await verifyMetaSignature(
        rawBody,
        request.headers.get("X-Hub-Signature-256"),
        config.metaAppSecret,
      );
      if (!signatureOk) return new Response("Unauthorized", { status: 401 });

      let payload: unknown;
      try {
        payload = JSON.parse(rawBody);
      } catch {
        return new Response("Bad request", { status: 400 });
      }

      for (const message of extractWhatsAppMessages(payload)) {
        if (message.from !== config.allowedWhatsAppNumber) continue;

        const key = `wa:${message.id}`;
        if (await env.DEDUP_KV.get(key)) continue;

        await env.DEDUP_KV.put(key, "1", { expirationTtl: DEDUP_TTL_SECONDS });
        ctx.waitUntil(
          processMessage(config, message, deps).catch((error) => {
            console.error("background message processing failed", {
              messageId: message.id,
              error: error instanceof Error ? error.message : "unknown error",
            });
          }),
        );
      }

      return new Response("EVENT_RECEIVED", { status: 200 });
    },
  } satisfies ExportedHandler<Env>;
}

export default createWorker();
```

Do not log `rawBody`, `message.text`, the configuration object, or upstream response bodies.
